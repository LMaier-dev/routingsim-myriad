# One eval run per line, executed from the repo's src/ directory.
# Blank lines and #-comments are ignored. run_evals.sh gives each line its own
# tmux window, so a run that hangs doesn't hold up the rest.
#
# --tag <name> suffixes the output dir; use it when two runs share a topology
# and scenario (e.g. basis=min vs tree) so they don't overwrite each other.
#
# Prefix a dynamics line with `SPLIT ` to run its GP and LS halves as parallel
# processes, then auto-combine the plots (--post). Same window, ~2x faster, and
# the flags stay written once. Example:
#   SPLIT python3 ../simulations/MYRIAD/eval_entrypoint.py dynamics --topo ../simulations/topologies/grid_10,10.txt --only churn --basis tree

# OVERVIEW: control-plane volume per phase (convergence, churn, each failure
# fraction) for every protocol variant, averaged over topologies AND failure
# schedules with the spread drawn as boxplots.
#   --tr-extremes  only the lowest-tr and highest-tr point of the ringgrow sweep,
#                  all 5 topology seeds of each -> 10 topologies, two far-apart
#                  points on the transition axis instead of the whole sweep
#   --repeat N     N failure schedules per topology (seed-set 1..N), so the
#                  boxes span both spreads; 10 topos x 2 scenarios x 4 configs
#                  x N seeds runs, all in one --jobs pool
#   no SPLIT       overview already fans the whole cross product out itself
# Windows: conv [0,--failure-start) is the initial RIB formation, then churn
# steady state / one window per node-wave fraction. churn/conv and node/conv
# measure the same phase in two different runs: they should agree.
python3 ../simulations/MYRIAD/eval_entrypoint.py dynamics --overview --dir ../simulations/MYRIAD/myriad_topologies/csv/delaunay_grow10000_A1200_spof0.5 --tr-extremes --basis native --repeat 10 --jobs 64 --no-refresh-flood --nospof --noreplay

# same, on the SPOF-free corpus: every transition two nodes wide, so a failure
# never severs a ring pair outright
python3 ../simulations/MYRIAD/eval_entrypoint.py dynamics --overview --dir ../simulations/MYRIAD/myriad_topologies/csv/delaunay_grow10000_A1200_spof0 --tr-extremes --basis native --repeat 10 --jobs 64 --no-refresh-flood --nospof --noreplay
