# One eval run per line, executed from the repo's src/ directory.
# Blank lines and #-comments are ignored. run_evals.sh gives each line its own
# tmux window, so a run that hangs doesn't hold up the rest.
#
# --tag <name> suffixes the output dir; use it when two runs share a topology
# and scenario (e.g. basis=min vs tree) so they don't overwrite each other.
#
# Prefix a dynamics line with `SPLIT ` to run its PV and LS halves as parallel
# processes, then auto-combine the plots (--post). Same window, ~2x faster, and
# the flags stay written once. Example:
#   SPLIT python3 ~/routingsim/simulations/MYRIAD/eval_entrypoint.py dynamics --topo ~/routingsim/simulations/topologies/grid_10,10.txt --only churn --basis tree
#
# --topo <dir>@first|median|last picks one *.txt from a swept family dir, sorted
# by _tr<N> (min/mid/max transition count), so a run names a representative
# without pinning a tr/fan/seed filename that regeneration renames.

# ---------------------------------------------------------------------------
# n=2500 corpus (generate_topos.sh grow). Same experiment as the n=10000 family
# -- rings 9.8..11.8 long, tr swept 300..700 -- at a quarter of the scale.
#
# Why quarter scale: memory is per-node RIB (every node holds a route to each
# origin ring), so it scales with n*A. One n=10000 PV run plateaued at 9.8GB;
# n=2500 lands around 1GB, well under 3GB even for LS. That is what lets all
# three lines below run at once: run_evals.sh has no concurrency gate, it opens
# every line's window together, and a SPLIT line starts one process PER CONFIG.
#   churn 4 configs (pv, pv_noreplay, pv_nospof, ls) + node 3 + overview 16
#   = 23 concurrent sims. At 10000 nodes that was 225GB; here it is ~25-70GB.
# If a per-run peak turns out higher than 3GB, --jobs on the overview line is
# the dial: it is 16 of those 23.
# ---------------------------------------------------------------------------

# detailed failure sequence, RIB snapshots ON: reconvergence latency per wave,
# route counts over time, stretch vs LS
SPLIT python3 ~/routingsim/simulations/MYRIAD/eval_entrypoint.py dynamics --topo ~/routingsim/simulations/MYRIAD/myriad_topologies/csv/delaunay_grow2500_A300_spof0.5@median --only churn --basis native --no-refresh-flood --nospof --noreplay

SPLIT python3 ~/routingsim/simulations/MYRIAD/eval_entrypoint.py dynamics --topo ~/routingsim/simulations/MYRIAD/myriad_topologies/csv/delaunay_grow2500_A300_spof0.5@median --only node --basis native --no-refresh-flood --noreplay

# overview: bytes and messages per phase (conv, churn, each wave fraction),
